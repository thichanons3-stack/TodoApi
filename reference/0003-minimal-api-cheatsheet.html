2 · กฎ parameter binding

ชนิด parameterที่มา
int id ที่มี {id} ใน routeRoute value
complex type ใน POST/PUT เช่น TodoItemDTO dtoBody (JSON)
ชนิดที่ลงทะเบียนใน DI เช่น TodoDb dbDI container (ไม่ต้องใช้ [FromServices])
primitive ที่ไม่มีใน route เช่น int pageQuery string
ระบุชัดเจน[FromRoute] [FromQuery] [FromBody] [FromHeader] [FromForm] [FromServices]
  • GET / HEAD / OPTIONS / DELETE ไม่อ่าน body อัตโนมัติ — ต้อง [FromBody] ถ้าจะใช้
  • parameter ธรรมดา = required — ถ้าไม่มีค่าให้ bind จะ error; อยากให้ optional ใช้ int? หรือค่าเริ่มต้น int page = 1
  • ชนิด route constraint เช่น {id:int} — ถ้าค่าไม่ตรง (เช่น /api/todoitems/abc) จะไม่จับคู่ → 404 (ทดสอบแล้ว)

4 · โครงไฟล์ที่ใช้จริงในโปรเจกต์นี้

// Endpoints/TodoItemsEndpoints.cs
public static class TodoItemsEndpoints
{
    public static void MapTodoItems(this WebApplication app)
    {
        var group = app.MapGroup("/api/todoitems");
        group.MapGet("/", GetTodoItems);
        group.MapGet("/{id:int}", GetTodoItem);
        group.MapPost("/", CreateTodoItem);
        group.MapPut("/{id:int}", UpdateTodoItem);
        group.MapDelete("/{id:int}", DeleteTodoItem);
    }

    private static async Task<IResult> GetTodoItems(TodoDb db)
    {
        var todos = await db.TodoItems.Include(t => t.Category).ToListAsync();
        return TypedResults.Ok(todos.Select(t => new TodoItemDTO(t)).ToList());
    }
    // handler อื่นๆ คัด logic จาก TodoItemsController ได้ทั้งดุ้น
}

// Program.cs — ครึ่งแรกไม่ต้องแก้ ครึ่งหลังเปลี่ยนแค่บรรทัดเดียว
TodoItemsEndpoints.MapTodoItems(app);   // แทนที่ app.MapControllers();

5 · กับดัก 3 อย่างที่ทดสอบกับ .NET 10 แล้ว

  • แย่งเส้นทาง — register สอง endpoint ที่ /api/todoitems เดียวกัน → request แรกได้ AmbiguousMatchException (HTTP 500) ตอนทดลองให้ใช้ prefix คนละอัน
  • validation ไม่วิ่งเอง — ไม่มี builder.Services.AddValidation(); → DataAnnotations บน DTO ถูกเพิกเฉย ข้อมูลผิดวิ่งเข้า handler เลย
  • MapGroup + MapGet("/") — จับทั้ง /api/todoitems และ /api/todoitems/ (มี/ไม่มี slash ท้าย ได้ 200 ทั้งคู่ — ทดสอบแล้ว)
